const BASE_URL = 'http://127.0.0'; let fotoArchivo = null; // 1. Efecto espejo: Escribir en tiempo real dentro de la credencial function vincularCampos() { const inputs = ['nombre', 'apellido1', 'apellido2', 'identificacion', 'brigada', 'sangre']; inputs.forEach(id => { document.getElementById(id).addEventListener('input', (e) => { let destinoId = 'cred' + id.charAt(0).toUpperCase() + id.slice(1); if (document.getElementById(destinoId)) { document.getElementById(destinoId).innerText = e.target.value || '-'; } }); }); } vincularCampos(); // 2. Evento para previsualizar la fotografía seleccionada document.getElementById('inputFoto').addEventListener('change', (e) => { const file = e.target.files[0]; if (file) { fotoArchivo = file; document.getElementById('credFoto').src = URL.createObjectURL(file); } }); // 3. Botón Buscar: Extrae datos existentes desde el servidor document.getElementById('btnBuscar').addEventListener('click', async () => { const id = document.getElementById('identificacion').value.trim().toUpperCase(); if (!id) return alert('Por favor, ingresa una identificación'); try { const res = await fetch(`${BASE_URL}/${id}`); const datos = await res.json(); if (datos.existe) { document.getElementById('estadoTitulo').innerText = "📝 Editando Perfil Existente"; const partes = datos.nombre.split(' '); document.getElementById('nombre').value = partes[0] || ''; document.getElementById('apellido1').value = partes[1] || ''; document.getElementById('apellido2').value = partes.slice(2).join(' ') || ''; document.getElementById('telefono').value = datos.telefono; document.getElementById('sangre').value = datos.sangre; document.getElementById('region').value = datos.region; document.getElementById('brigada').value = datos.brigada; document.getElementById('credFoto').src = datos.foto_url; document.getElementById('credNombre').innerText = partes[0] || '-'; document.getElementById('credApellido1').innerText = partes[1] || '-'; document.getElementById('credApellido2').innerText = partes.slice(2).join(' ') || '-'; document.getElementById('credIdentificacion').innerText = id; document.getElementById('credBrigada').innerText = datos.brigada; document.getElementById('credSangre').innerText = datos.sangre; document.getElementById('btnGuardar').innerText = "💾 Actualizar Datos"; } else { document.getElementById('estadoTitulo').innerText = "✨ Nuevo Registro"; document.getElementById('perfilForm').reset(); document.getElementById('identificacion').value = id; document.getElementById('credFoto').src = 'https://placeholder.com'; document.getElementById('btnGuardar').innerText = "💾 Registrar como Nuevo"; } } catch (err) { alert('Error al conectar con la base de datos'); } }); // 4. Formulario Submit: Construye el paquete y lo manda al Servidor document.getElementById('perfilForm').addEventListener('submit', async (e) => { e.preventDefault(); const nombreCompleto = `${document.getElementById('nombre').value} ${document.getElementById('apellido1').value} ${document.getElementById('apellido2').value}`.trim(); const formData = new FormData(); formData.append('identificacion', document.getElementById('identificacion').value.trim().toUpperCase()); formData.append('nombre', nombreCompleto); formData.append('telefono', document.getElementById('telefono').value); formData.append('sangre', document.getElementById('sangre').value); formData.append('region', document.getElementById('region').value); formData.append('brigada', document.getElementById('brigada').value); if (fotoArchivo) formData.append('foto', fotoArchivo); try { const res = await fetch(`${BASE_URL}/guardar`, { method: 'POST', body: formData }); const resultado = await res.json(); if (resultado.status === 'success') { alert('🎉 ¡Perfil guardado con éxito!'); document.getElementById('credFoto').src = resultado.foto_url; fotoArchivo = null; } } catch (err) { alert('🔴 Error al conectar con el servidor'); } });
